\chapter{List of Symbols}\label{chap:symbols}

\begin{longtable}{L{4.2cm}L{9.8cm}}
    \toprule
    Symbol & Meaning \\
    \midrule
    \endfirsthead
    \toprule
    Symbol & Meaning \\
    \midrule
    \endhead
    $x, x^*$ & Candidate solution and global optimum \\
    $D$ & Dimension of the continuous search space \\
    $\mathrm{lb},\mathrm{ub}$ & Lower and upper search bounds \\
    $f_{\mathrm{clean}}(x), f^*$ & Clean objective value and its global minimum \\
    $f_{\mathrm{obs}}(x)$ & Objective value observed by the optimizer \\
    $\sigma$ & Relative noise intensity \\
    $\sigma_{\mathrm{synth}},\sigma_{\mathrm{eval}}$ & Noise intensities during synthesis and independent evaluation \\
    $s_{\mathrm{landscape}}$ & Mean clean optimality gap used to calibrate the alternative constant-noise model \\
    $\sigma_{\mathrm{hom}}$ & Standard deviation of the alternative constant-noise model \\
    $B_{\mathrm{eval}}$ & Objective-evaluation budget in independent benchmarking \\
    $n_r$ & Number of actual objective evaluations in benchmark trial $r$ \\
    $g_r(b)$ & Best clean optimality gap found by trial $r$ within $b$ evaluations \\
    $\tau$ & Clean optimality-gap target \\
    $T_{r,\tau}$ & First evaluation count at which trial $r$ reaches target $\tau$ \\
    $\mathcal{R},\mathcal{T}$ & Trial set and target set used for an attainment curve \\
    $\widehat F_{\mathcal R,\mathcal T}(b)$ & Fraction of trial--target pairs attained within $b$ evaluations \\
    \bottomrule
\end{longtable}
