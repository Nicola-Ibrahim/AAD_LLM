\begin{tikzpicture}[
    font=\footnotesize,
    >=Stealth,
    ioNode/.style={
        rectangle,
        draw=black!80,
        fill=yellow!15,
        text width=12.2cm,
        align=center,
        rounded corners=4pt,
        inner sep=7pt,
        thick
    },
    processNode/.style={
        rectangle,
        draw=black!80,
        fill=blue!5,
        text width=5.6cm,
        align=center,
        rounded corners=4pt,
        inner sep=6pt,
        thick,
        minimum height=1.3cm
    },
    evalNode/.style={
        processNode,
        fill=blue!10
    },
    selectionNode/.style={
        rectangle,
        draw=black!80,
        fill=purple!10,
        text width=12.2cm,
        align=center,
        rounded corners=4pt,
        inner sep=7pt,
        thick,
        minimum height=1.35cm
    },
    decision/.style={
        diamond,
        draw=black!80,
        fill=gray!15,
        thick,
        aspect=2.8,
        inner sep=3pt,
        align=center,
        font=\scriptsize,
        minimum width=5.6cm
    },
    outNode/.style={
        rectangle,
        draw=black!80,
        fill=green!15,
        text width=12.2cm,
        align=center,
        rounded corners=4pt,
        inner sep=7pt,
        thick,
        font=\footnotesize\bfseries
    },
    arrow/.style={
        ->,
        >=Stealth,
        thick,
        draw=black!75
    },
    fbArrow/.style={
        ->,
        >=Stealth,
        thick,
        draw=purple!80!black
    },
    elblB/.style={
        font=\scriptsize\bfseries,
        fill=white,
        inner sep=2pt,
        rounded corners=1pt
    }
]

    %----------------------------------------------------------------------
    % 1. Input Interface (Fed directly from Macro Pipeline)
    %----------------------------------------------------------------------
    \node[ioNode] (inputs) {
        \textbf{LLaMEA Synthesis Input}\\[2pt]
        \scriptsize Target Instance $(f_p, D, \sigma)$ \quad $\bullet$ \quad
        Active Prompt Scaffold $\mathcal{P}_{i,j}$ \quad $\bullet$ \quad
        Synthesis Budget $B_{\text{synth}} = 1{,}000$
    };

    %----------------------------------------------------------------------
    % 2. Evolutionary Context & Prompt Compilation
    %----------------------------------------------------------------------
    \node[
        processNode,
        below=0.75cm of inputs
    ] (context) {
        \textbf{Reflective Context Assembly}\\[2pt]
        \scriptsize Compiles $\mathcal{C}_g = \{\mathcal{P}_{i,j}, \mathcal{A}_{\text{parent}}, \mathcal{T}_{\text{feedback}}, \mathcal{I}_{\text{directive}}\}$
    };

    %----------------------------------------------------------------------
    % 3. Foundation Model Code Mutation
    %----------------------------------------------------------------------
    \node[
        processNode,
        below=0.75cm of context
    ] (generation) {
        \textbf{LLM Candidate Mutation}\\[2pt]
        \scriptsize Generates executable optimization heuristic script $\mathcal{A}_{\text{child}}$
    };

    %----------------------------------------------------------------------
    % 4. AST Validation and Security Containment
    %----------------------------------------------------------------------
    \node[
        processNode,
        below=0.75cm of generation
    ] (validation) {
        \textbf{AST Verification \& Parsing}\\[2pt]
        \scriptsize Screens imports, whitelists syntax, and extracts candidate class
    };

    %----------------------------------------------------------------------
    % 5. Sandboxed Runtime Execution
    %----------------------------------------------------------------------
    \node[
        processNode,
        below=0.75cm of validation
    ] (execution) {
        \textbf{Isolated Sandboxed Execution}\\[2pt]
        \scriptsize Executes candidate within process isolate ($T_{\text{max}} = 30.0\,\text{s}$)
    };

    %----------------------------------------------------------------------
    % 6. Dual-Decoupled In-Loop Evaluation Channels
    %----------------------------------------------------------------------
    \node[
        evalNode,
        below left=0.85cm and 0.25cm of execution.south,
        anchor=north east
    ] (stochastic) {
        \textbf{Stochastic Search Execution}\\[2pt]
        \scriptsize Heuristic searches strictly via noisy evaluations $f_{\text{noisy}}(x_k)$
    };

    \node[
        evalNode,
        below right=0.85cm and 0.25cm of execution.south,
        anchor=north west
    ] (clean) {
        \textbf{Clean Ground-Truth Oracle}\\[2pt]
        \scriptsize Evaluates terminal coordinate on true objective: $y_{\text{clean}} = f_{\text{clean}}(\mathbf{x}^*)$
    };

    %----------------------------------------------------------------------
    % 7. Selection, Resampling, and Context Update
    %----------------------------------------------------------------------
    \coordinate (evalCenter) at ($(stochastic.south)!0.5!(clean.south)$);

    \node[
        selectionNode,
        below=0.85cm of evalCenter
    ] (selection) {
        \textbf{Parent--Child Comparison \& Incumbent Resampling}\\[2pt]
        \scriptsize Evaluates selection gradient $\mathcal{F} = -|y_{\text{clean}} - f^*|$, resamples incumbent parent ($K=3$),\\
        and appends diagnostic feedback buffer $\mathcal{T}_{\text{feedback}}$
    };

    %----------------------------------------------------------------------
    % 8. Termination Check
    %----------------------------------------------------------------------
    \node[
        decision,
        below=0.85cm of selection
    ] (termination) {
        \textbf{Generational Budget Reached?} ($g = G$)
    };

    %----------------------------------------------------------------------
    % 9. Terminal Output
    %----------------------------------------------------------------------
    \node[
        outNode,
        below=0.85cm of termination
    ] (output) {
        \textbf{Synthesized Champion Heuristic ($\mathcal{A}^*$)}\\[2pt]
        \scriptsize Validated Python optimizer class script and generational synthesis telemetry
    };

    %----------------------------------------------------------------------
    % Background Enclosure (Iterative Loop Boundary)
    %----------------------------------------------------------------------
    \begin{scope}[on background layer]
        \node[
            draw=gray!50,
            dashed,
            fill=gray!3,
            rounded corners=8pt,
            inner sep=12pt,
            fit=(context) (generation) (validation) (execution)
                 (stochastic) (clean) (selection),
            label={[font=\bfseries\scriptsize, text=gray!70!black]above:Iterative LLaMEA Evolutionary Engine}
        ] (loopGroup) {};
    \end{scope}

    %----------------------------------------------------------------------
    % Forward Connectors
    %----------------------------------------------------------------------
    \draw[arrow] (inputs) -- (context);
    \draw[arrow] (context) -- (generation);
    \draw[arrow] (generation) -- node[elblB, right=2pt] {$\mathcal{A}_{\text{child}}$ Script} (validation);
    \draw[arrow] (validation) -- (execution);

    % Dual Evaluation Forks
    \draw[arrow] (execution.south) -- ++(0,-0.3) -| (stochastic.north);
    \draw[arrow] (execution.south) -- ++(0,-0.3) -| (clean.north);

    % Dual Evaluation Convergence
    \draw[arrow] (stochastic.south) -- ++(0,-0.3) -| ([xshift=-2.8cm]selection.north);
    \draw[arrow] (clean.south)      -- ++(0,-0.3) -| ([xshift=2.8cm]selection.north);

    % Termination Pathways
    \draw[arrow] (selection) -- (termination);
    \draw[arrow] (termination) -- node[elblB, right=2pt] {Yes} (output);

    % Outer Loop Feedback (Routes around the outside left edge)
    \draw[fbArrow] (termination.west) -- ++(-3.9,0)
        node[elblB, pos=0.5, above, text=purple!80!black] {No: Next Generation ($g \gets g+1$)}
        |- (context.west);

\end{tikzpicture}
