\begin{tikzpicture}[
    font=\footnotesize,
    >=Stealth,
    ioNode/.style={
        rectangle,
        draw=black!80,
        fill=yellow!15,
        text width=13cm,
        align=center,
        rounded corners=4pt,
        inner sep=8pt,
        thick
    },
    processNode/.style={
        rectangle,
        draw=black!80,
        fill=blue!5,
        text width=5.7cm,
        align=center,
        rounded corners=4pt,
        inner sep=7pt,
        thick,
        minimum height=1.45cm
    },
    evalNode/.style={
        processNode,
        fill=blue!10
    },
    selectionNode/.style={
        rectangle,
        draw=black!80,
        fill=purple!10,
        text width=12.5cm,
        align=center,
        rounded corners=4pt,
        inner sep=8pt,
        thick,
        minimum height=1.6cm
    },
    decision/.style={
        diamond,
        draw=black!80,
        fill=gray!15,
        thick,
        aspect=3.0,
        inner sep=4pt,
        align=center,
        font=\scriptsize,
        minimum width=5.8cm
    },
    outNode/.style={
        rectangle,
        draw=black!80,
        fill=green!15,
        text width=13cm,
        align=center,
        rounded corners=4pt,
        inner sep=8pt,
        thick,
        font=\footnotesize\bfseries
    },
    arrow/.style={
        ->,
        >=Stealth,
        thick,
        draw=black!75
    },
    fbArrow/.style={
        ->,
        >=Stealth,
        thick,
        draw=purple!80!black
    },
    elbl/.style={
        font=\scriptsize,
        fill=white,
        inner sep=2pt,
        rounded corners=1pt
    },
    elblB/.style={
        elbl,
        font=\scriptsize\bfseries
    }
]

%----------------------------------------------------------------------
% Input
%----------------------------------------------------------------------

\node[ioNode] (inputs) {
    \textbf{Optimization Task and Synthesis Configuration} \\[3pt]
    \scriptsize
    Continuous black-box problem
    \quad $\bullet$ \quad
    Prompt configuration
    \quad $\bullet$ \quad
    Synthesis budget
};

%----------------------------------------------------------------------
% Synthesis context
%----------------------------------------------------------------------

\node[
    processNode,
    below=0.8cm of inputs
] (context) {
    \textbf{Synthesis Context} \\[3pt]
    \scriptsize
    Optimization task, previous candidate information,
    incumbent, and prompt configuration
};

%----------------------------------------------------------------------
% Generation
%----------------------------------------------------------------------

\node[
    processNode,
    below=0.8cm of context
] (generation) {
    \textbf{LLM Candidate Generation} \\[3pt]
    \scriptsize
    Generates executable optimization heuristic
    $\mathcal{A}_{\text{child}}$
};

%----------------------------------------------------------------------
% Code extraction and validation
%----------------------------------------------------------------------

\node[
    processNode,
    below=0.8cm of generation
] (validation) {
    \textbf{Code Extraction and Validation} \\[3pt]
    \scriptsize
    Extracts generated source code and checks
    syntactic and structural constraints
};

%----------------------------------------------------------------------
% Execution
%----------------------------------------------------------------------

\node[
    processNode,
    below=0.8cm of validation
] (execution) {
    \textbf{Isolated Candidate Execution} \\[3pt]
    \scriptsize
    Runs the candidate under a controlled execution
    environment and timeout
};

%----------------------------------------------------------------------
% Evaluation: two parallel components
%----------------------------------------------------------------------

\node[
    evalNode,
    below left=1.0cm and 0cm of execution.south,
    anchor=north west
] (stochastic) {
    \textbf{Stochastic Objective Evaluation} \\[3pt]
    \scriptsize
    Candidate interacts with the objective through
    noisy fitness observations
};

\node[
    evalNode,
    right=1.0cm of stochastic
] (clean) {
    \textbf{Clean Diagnostic Evaluation} \\[3pt]
    \scriptsize
    Returned solution is assessed using the
    underlying clean objective
};

%----------------------------------------------------------------------
% Selection and feedback
%----------------------------------------------------------------------

\node[
    selectionNode,
    below=1.0cm of $(stochastic.south)!0.5!(clean.south)$
] (selection) {
    \textbf{Parent--Child Comparison and Context Update} \\[3pt]
    \scriptsize
    Compares candidate performance with the incumbent,
    updates the retained solution, and constructs feedback
    for the next generation
};

%----------------------------------------------------------------------
% Termination
%----------------------------------------------------------------------

\node[
    decision,
    below=1.0cm of selection
] (termination) {
    \textbf{Synthesis Terminated?}
};

%----------------------------------------------------------------------
% Output
%----------------------------------------------------------------------

\node[
    outNode,
    below=1.0cm of termination
] (output) {
    \textbf{Final Output: Synthesized Optimization Heuristic}
    \\[3pt]
    \scriptsize
    Retained incumbent algorithm and associated synthesis information
};

%----------------------------------------------------------------------
% Loop boundary
%----------------------------------------------------------------------

\begin{scope}[on background layer]
    \node[
        draw=gray!50,
        dashed,
        fill=gray!3,
        rounded corners=8pt,
        inner sep=12pt,
        fit=(context) (generation) (validation) (execution)
             (stochastic) (clean) (selection),
        label={[
            font=\bfseries\scriptsize,
            gray!70!black
        ]above:Iterative LLaMEA Synthesis Loop}
    ] (loopGroup) {};
\end{scope}

%----------------------------------------------------------------------
% Forward arrows
%----------------------------------------------------------------------

\draw[arrow]
    (inputs.south)
    -- (context.north);

\draw[arrow]
    (context.south)
    -- (generation.north);

\draw[arrow]
    (generation.south)
    -- node[elblB, right=2pt]
    {Candidate program}
    (validation.north);

\draw[arrow]
    (validation.south)
    -- (execution.north);

%----------------------------------------------------------------------
% Evaluation arrows
%----------------------------------------------------------------------

\draw[arrow]
    (execution.south)
    -- ++(0,-0.35)
    -| (stochastic.north);

\draw[arrow]
    (execution.south)
    -- ++(0,-0.35)
    -| (clean.north);

\draw[arrow]
    (stochastic.south)
    -- ++(0,-0.45)
    -| (selection.west);

\draw[arrow]
    (clean.south)
    -- ++(0,-0.45)
    -| (selection.east);

%----------------------------------------------------------------------
% Termination
%----------------------------------------------------------------------

\draw[arrow]
    (selection.south)
    -- (termination.north);

\draw[arrow]
    (termination.south)
    -- node[elblB, right=2pt]
    {Yes}
    (output.north);

%----------------------------------------------------------------------
% Next-generation feedback
%----------------------------------------------------------------------

\draw[fbArrow]
    (termination.west)
    -- ++(-2.8,0)
    node[
        elblB,
        pos=0.5,
        above,
        text=purple!80!black
    ] {No: next generation}
    |- ($(context.west)+(-0.6,0)$);

\end{tikzpicture}
