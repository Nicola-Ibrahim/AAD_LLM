\begin{tikzpicture}[
    font=\footnotesize,
    >=Stealth,
    ioNode/.style={
        rectangle,
        draw=black!80,
        fill=yellow!15,
        text width=12.2cm,
        align=center,
        rounded corners=4pt,
        inner sep=5.5pt,
        thick
    },
    processNode/.style={
        rectangle,
        draw=black!80,
        fill=blue!5,
        text width=9.2cm,
        align=center,
        rounded corners=4pt,
        inner sep=5pt,
        thick,
        minimum height=0.95cm
    },
    evalNode/.style={
        processNode,
        fill=blue!10
    },
    decision/.style={
        diamond,
        draw=black!80,
        fill=gray!15,
        thick,
        aspect=2.3,
        inner sep=2pt,
        align=center,
        font=\scriptsize
    },
    actionBox/.style={
        rectangle,
        draw=black!75,
        thick,
        rounded corners=3pt,
        align=center,
        inner sep=4pt,
        font=\scriptsize,
        text width=4.5cm,
        minimum height=0.85cm
    },
    mergeNode/.style={
        rectangle,
        draw=black!80,
        fill=purple!10,
        text width=12.2cm,
        align=center,
        rounded corners=4pt,
        inner sep=5pt,
        thick
    },
    outNode/.style={
        rectangle,
        draw=black!80,
        fill=green!15,
        text width=12.2cm,
        align=center,
        rounded corners=4pt,
        inner sep=5.5pt,
        thick,
        font=\footnotesize\bfseries
    },
    arrow/.style={
        ->,
        >=Stealth,
        thick,
        draw=black!75
    },
    fbArrow/.style={
        ->,
        >=Stealth,
        thick,
        draw=purple!80!black
    },
    errArrow/.style={
        ->,
        >=Stealth,
        thick,
        draw=red!75!black
    },
    elblB/.style={
        font=\scriptsize\bfseries,
        fill=white,
        inner sep=1.5pt,
        rounded corners=1pt
    }
]

    %----------------------------------------------------------------------
    % 1. Input Specification
    %----------------------------------------------------------------------
    \node[ioNode] (inputs) {
        \textbf{LLaMEA Synthesis Input}\\[1pt]
        \scriptsize Target Problem $(f_p, D, \sigma)$ \quad $\bullet$ \quad
        Scaffold Prompt Template $\mathcal{P}_{i,j}$ \quad $\bullet$ \quad
        Inner Search Budget $B_{\text{synth}} = 1e^6$ evaluations
    };

    %----------------------------------------------------------------------
    % 2. Dynamic Context Assembly
    %----------------------------------------------------------------------
    \node[
        processNode,
        below=0.45cm of inputs
    ] (context) {
        \textbf{Reflective Context Assembly}\\[1pt]
        \scriptsize Builds $\mathcal{C}_g = \{\mathcal{P}_{\text{system}}, \mathcal{H}_{1 \dots g-1}, \mathcal{T}_{\text{feedback}}, \mathcal{I}_{\text{directive}}\}$\\
    };

    %----------------------------------------------------------------------
    % 3. LLM Semantic Mutation Operator
    %----------------------------------------------------------------------
    \node[
        processNode,
        fill=blue!8,
        below=0.45cm of context
    ] (generation) {
        \textbf{LLM Candidate Mutation Operator}\\[1pt]
        \scriptsize Mutates code to produce offspring script $\mathcal{A}_{\text{child}}$\\
    };

    %----------------------------------------------------------------------
    % 4. AST Validation & Isolated Sandboxed Execution
    %----------------------------------------------------------------------
    \node[
        evalNode,
        below=0.45cm of generation
    ] (execSandbox) {
        \textbf{AST Verification \& Sandboxed Runtime Execution}\\[1pt]
        \scriptsize Run $\mathcal{A}_{\text{child}}$ against noisy target $f_{\text{noisy}}(x_k)$ ($B_{\text{synth}} = 1{,}000$, $T_{\text{max}} = 30.0\,\text{s}$)
    };

    %----------------------------------------------------------------------
    % 5. Validation & Execution Gate
    %----------------------------------------------------------------------
    \node[
        decision,
        text width=3.8cm,
        below=0.45cm of execSandbox
    ] (checkExecution) {
        \textbf{Execution Successful?}
    };

    % 5b. Fault Interception Block (Right Bypass)
    \node[
        actionBox,
        fill=red!8,
        draw=red!65!black,
        right=3cm of checkExecution,
        text width=3.4cm
    ] (execFail) {
        \textbf{Fault Interception}\\[1pt]
        \tiny Assign tiered penalty $\mathcal{F}_{\text{failure}}$\\
        \tiny Extract traceback \& error line\\
        \tiny Format diagnostic self-repair directive\\
        \tiny Retain active incumbent $\mathcal{A}_{\text{parent}}$
    };

    %----------------------------------------------------------------------
    % 6. Clean Ground-Truth Evaluation
    %----------------------------------------------------------------------
    \node[
        evalNode,
        fill=green!8,
        draw=green!55!black,
        below=0.55cm of checkExecution
    ] (cleanOracle) {
        \textbf{Clean Ground-Truth Evaluation}\\[1pt]
        \scriptsize Evaluate best coordinate: $y_{\text{clean}} = f_{\text{clean}}(\mathbf{x}^*)$\\
        \scriptsize Optimality Gap: $\Delta y = |y_{\text{clean}} - f^*| \implies \text{Fitness } \mathcal{F}_{\text{child}} = -\Delta y$
    };

    %----------------------------------------------------------------------
    % 7. (1+1) Incumbent Selection Gate
    %----------------------------------------------------------------------
    \node[
        decision,
        text width=3.8cm,
        below=0.45cm of cleanOracle
    ] (selectDecision) {
        \textbf{Child Better than Parent?}\\[1pt]
        \tiny ($\mathcal{F}_{\text{child}} > \mathcal{F}_{\text{parent}}$)
    };

    % 7a. Accept Child (Left)
    \node[
        actionBox,
        fill=green!10,
        draw=green!60!black,
        below left=0.4cm and 0.2cm of selectDecision.south,
        anchor=north east
    ] (updateIncumbent) {
        \textbf{Accept Child \& Update}\\[1pt]
        $\mathcal{A}_{\text{parent}} \gets \mathcal{A}_{\text{child}}$, \quad $\mathcal{F}_{\text{parent}} \gets \mathcal{F}_{\text{child}}$\\
        \tiny Reset failure counter ($C_{\text{failures}} \gets 0$)
    };

    % 7b. Reject Child (Right)
    \node[
        actionBox,
        fill=red!10,
        draw=red!60!black,
        below right=0.4cm and 0.2cm of selectDecision.south,
        anchor=north west
    ] (retainIncumbent) {
        \textbf{Reject Child \& Retain}\\[1pt]
        $\mathcal{A}_{\text{parent}}$ remains unchanged\\
        \tiny Increment failure counter ($C_{\text{failures}} \gets C_{\text{failures}} + 1$)
    };

    %----------------------------------------------------------------------
    % 8. Synchronization & Context Log Update
    %----------------------------------------------------------------------
    \coordinate (actionMid) at ($(updateIncumbent.south)!0.5!(retainIncumbent.south)$);

    \node[
        mergeNode,
        below=0.45cm of actionMid
    ] (syncContext) {
        \textbf{Generational Context \& Telemetry Synchronization}\\[1pt]
        \scriptsize Append run record to $\mathcal{H}_{1 \dots g-1}$, persist artifacts to database, and compile feedback buffer $\mathcal{T}_{\text{feedback}}$
    };

    %----------------------------------------------------------------------
    % 9. Termination Check
    %----------------------------------------------------------------------
    \node[
        decision,
        text width=4.2cm,
        below=0.45cm of syncContext
    ] (termination) {
        \textbf{Evolution Finished?}\\[1pt]
        \tiny Generation $g = 10$ Reached
    };

    %----------------------------------------------------------------------
    % 10. Final Terminal Output
    %----------------------------------------------------------------------
    \node[
        outNode,
        below=0.45cm of termination
    ] (output) {
        \textbf{Synthesized Champion Heuristic ($\mathcal{A}^*$)}\\[1pt]
        \scriptsize Best retained algorithm script and complete 10-generation synthesis telemetry
    };

    %----------------------------------------------------------------------
    % Background Container Enclosure
    %----------------------------------------------------------------------
    \begin{scope}[on background layer]
        \node[
            draw=gray!50,
            dashed,
            fill=gray!3,
            rounded corners=8pt,
            inner sep=10pt,
            fit=(context) (generation) (execSandbox) (checkExecution) 
                 (execFail) (cleanOracle) (selectDecision) (updateIncumbent) 
                 (retainIncumbent) (syncContext),
            label={[font=\bfseries\scriptsize, text=gray!70!black]above:Iterative LLaMEA Synthesis Loop ($10$ Generations)}
        ] (loopGroup) {};
    \end{scope}

    %----------------------------------------------------------------------
    % Forward Connectors
    %----------------------------------------------------------------------
    \draw[arrow] (inputs) -- (context);
    \draw[arrow] (context) -- (generation);
    \draw[arrow] (generation) -- node[elblB, right=2pt] {$\mathcal{A}_{\text{child}}$ Script} (execSandbox);
    \draw[arrow] (execSandbox) -- (checkExecution);

    % Post-Execution Gate Branches
    \draw[arrow] (checkExecution.south) -- node[elblB, right=2pt] {Yes: Valid Run} (cleanOracle);
    \draw[errArrow] (checkExecution.east) -- node[elblB, above=1pt] {No: AST/Runtime Fault} (execFail.west);

    % Ground Truth to Selection
    \draw[arrow] (cleanOracle) -- node[elblB, right=2pt] {$\mathcal{F}_{\text{child}} = -\Delta y$} (selectDecision);

    % Selection Gate Branches
    \draw[arrow] (selectDecision.south) -| node[elblB, pos=0.25, above] {Yes} (updateIncumbent.north);
    \draw[arrow] (selectDecision.south) -| node[elblB, pos=0.25, above] {No} (retainIncumbent.north);

    % Converge into Synchronization Node
    \draw[arrow] (updateIncumbent.south) -- ++(0,-0.2) -| ([xshift=-2.8cm]syncContext.north);
    \draw[arrow] (retainIncumbent.south) -- ++(0,-0.2) -| ([xshift=2.8cm]syncContext.north);
    
    % Fault Pathway to Synchronization (Bypasses clean oracle & selection)
    \draw[errArrow] (execFail.south) -- ++(0,-3.45) -| ([xshift=5.2cm]syncContext.north);

    % Termination Pathways
    \draw[arrow] (syncContext) -- (termination);
    \draw[arrow] (termination) -- node[elblB, right=2pt] {Yes ($g = 10$)} (output);

    % Outer Loop Feedback (Routes to Context Assembly for Next Mutation)
    \draw[fbArrow] (termination.west) -- ++(-3.6,0)
        node[elblB, pos=0.5, above, text=purple!80!black] {No: Next Iteration ($g \gets g+1$)}
        |- (context.west);

\end{tikzpicture}
