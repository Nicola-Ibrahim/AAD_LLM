\begin{tikzpicture}[
    node distance=1.1cm,
    font=\small\sffamily,
    >=Stealth,
    % --- Node Styles ---
    ioNode/.style={
        rectangle, 
        rounded corners=4pt, 
        draw=gray!70!black, 
        fill=gray!10, 
        thick, 
        minimum width=7.2cm, 
        minimum height=1.1cm, 
        align=center
    },
    stageNode/.style={
        rectangle, 
        rounded corners=5pt, 
        draw=#1!80!black, 
        fill=#1!8, 
        thick, 
        minimum width=7.2cm, 
        minimum height=1.2cm, 
        align=center
    },
    outNode/.style={
        rectangle, 
        rounded corners=4pt, 
        draw=orange!80!black, 
        fill=yellow!20, 
        thick, 
        minimum width=7.2cm, 
        minimum height=1.1cm, 
        align=center
    },
    % --- Path Styles ---
    mainArrow/.style={
        ->, 
        very thick, 
        draw=gray!75!black
    },
    feedbackArrow/.style={
        ->, 
        very thick, 
        draw=blue!80!black, 
        dashed
    }
]

    % =========================================================================
    % VERTICAL PIPELINE NODES
    % =========================================================================

    % System Input
    \node (input) [ioNode] {
        \textbf{Problem Specification}\\[-1pt]
        \footnotesize Problem Definition, Search Bounds, and Interface Template
    };

    % Stage 1: LLM Engine
    \node (llm) [stageNode=blue, below=1.0cm of input] {
        \textbf{Stage 1: Generative Model}\\[-1pt]
        \footnotesize Synthesizes and Modifies Optimization Code
    };

    % Stage 2: Sandbox
    \node (sandbox) [stageNode=red, below=1.0cm of llm] {
        \textbf{Stage 2: Execution Sandbox}\\[-1pt]
        \footnotesize Runs Candidate Algorithm on Target Functions
    };

    % Stage 3: Evaluation Module
    \node (eval) [stageNode=green!70!black, below=1.0cm of sandbox] {
        \textbf{Stage 3: Performance Assessment}\\[-1pt]
        \footnotesize Computes Target Metrics and Logs Runtime Feedback
    };

    % System Output
    \node (output) [outNode, below=1.0cm of eval] {
        \textbf{Discovered Optimization Algorithm}\\[-1pt]
        \footnotesize Final Evolved Search Heuristic
    };

    % =========================================================================
    % FORWARD ARROWS (NO LABELS)
    % =========================================================================

    \draw [mainArrow] (input) -- (llm);
    \draw [mainArrow] (llm) -- (sandbox);
    \draw [mainArrow] (sandbox) -- (eval);
    \draw [mainArrow] (eval) -- (output);

    % =========================================================================
    % EVOLUTIONARY FEEDBACK LOOP
    % =========================================================================

    \draw [feedbackArrow] (eval.west) 
        .. controls ($(eval.west)+(-3.2,0)$) and ($(llm.west)+(-3.2,0)$) .. 
        node[left=6pt, align=center, text=blue!80!black, font=\footnotesize\sffamily] {
            \textbf{Evolutionary Feedback Loop}\\[2pt]
            Generations $1 \dots N-1$\\[3pt]
            Performance Metrics \&\\
            Execution Logs
        } (llm.west);

    % =========================================================================
    % OUTER CONTAINER
    % =========================================================================
    \node[
        draw=gray!40, 
        dashed, 
        inner sep=0.4cm, 
        rounded corners=8pt, 
        fit=(llm) (sandbox) (eval)
    ] {};

\end{tikzpicture}
