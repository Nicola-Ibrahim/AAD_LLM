\begin{tikzpicture}[
    font=\footnotesize,
    >=Stealth,
    % ---------------------------------------------------------------
    % Global Styling & Dimensions
    % ---------------------------------------------------------------
    baseNode/.style={
        rectangle,
        draw=black!80,
        thick,
        rounded corners=4pt,
        align=center,
        inner sep=7pt
    },
    wideNode/.style={
        baseNode,
        text width=12.2cm
    },
    stageNode/.style={
        baseNode,
        fill=blue!5,
        text width=12.2cm,
        minimum height=1.15cm
    },
    evalBranch/.style={
        baseNode,
        fill=blue!10,
        text width=5.6cm,
        minimum height=1.15cm
    },
    arrow/.style={
        ->,
        >=Stealth,
        thick,
        draw=black!75
    },
    branchArrow/.style={
        ->,
        >=Stealth,
        thick,
        draw=black!65
    }
]

    % ==================================================================
    % 1. EXPERIMENTAL INPUT & TARGET SETUP
    % ==================================================================
    \node[wideNode, fill=yellow!15] (problem) {
        \textbf{Problem and Setup} \\[2pt]
        \scriptsize
        $\bullet$ \quad
        BBOB benchmark problems \quad $\bullet$ \quad
        Evaluation budget $B$ \\
        noise injection to corrupt objective fitness observations
    };

    % ==================================================================
    % 2. UNIFIED PROMPT SCAFFOLDING BLOCK
    % ==================================================================
    \node[
        wideNode,
        fill=orange!10,
        draw=orange!80!black,
        below=0.85cm of problem
    ] (promptScaffolding) {
        \textbf{Prompt Scaffolding Framework} \\[2pt]
        \scriptsize
        Invariant Prompts (API contract, defensive format, budget limits) \\
        coupled with Variant Prompts: Information Modes ($\textsc{Explicit}, \textsc{Implicit}$)
        $\times$ 4 Search Strategy Scaffolds
    };

    % ==================================================================
    % 3. LLAMEA SYNTHESIS & INTERNAL EVALUATOR / EXECUTOR
    % ==================================================================
    \node[
        stageNode,
        below=0.85cm of promptScaffolding
    ] (synthesis) {
        \textbf{LLaMEA Generative Synthesis Engine} \\[2pt]
        \scriptsize
        Iterative $(1+1)$ mutation loop: passes candidate heuristic $\mathcal{A}_{\text{child}}$ to internal \textbf{\texttt{Evaluator}} \\
        which drives the sandboxed \textbf{\texttt{AlgorithmExecutor}} to capture execution traces and clean ground-truth fitness
    };

    % ==================================================================
    % 4. SYNTHESIS OUTPUT
    % ==================================================================
    \node[
        stageNode,
        below=0.8cm of synthesis
    ] (champions) {
        \textbf{Synthesized Algorithm Champions ($\mathcal{A}^*$)} \\[2pt]
        \scriptsize
        Retained candidate optimization algorithms and corresponding generational telemetry
    };

    % ==================================================================
    % 5. EXPERIMENTAL ARTIFACT QUERY & CONFIGURATION RESOLUTION
    % ==================================================================
    \node[
        stageNode,
        below=0.85cm of champions
    ] (benchmarkConfig) {
        \textbf{Benchmark Configuration \& Registry Query} \\[2pt]
        \scriptsize
        Queries persisted synthesis records to extract algorithm code and their exact training configurations \\
        (synthesis budget $B_{\text{synth}}$, injected noise level $\sigma$, prompt scaffolding archetype, and model scale)
    };

    % ==================================================================
    % 6. DUAL BENCHMARK EVALUATION REGIMES
    % ==================================================================
    \node[
        evalBranch,
        below left=0.75cm and 0.25cm of benchmarkConfig.south,
        anchor=north east
    ] (evalSynthesized) {
        \textbf{Multi-Seed Re-evaluation} \\[2pt]
        \scriptsize
        Executes queried algorithm configurations across multiple independent test seeds \\
        under target operational environments
    };

    \node[
        evalBranch,
        below right=0.75cm and 0.25cm of benchmarkConfig.south,
        anchor=north west
    ] (evalBaselines) {
        \textbf{Baseline Comparison Suite} \\[2pt]
        \scriptsize
        Parallel evaluation of classical reference optimizers \\
        (CMA-ES, DE, PSO) on identical benchmark problem instances
    };

    % ==================================================================
    % 7. STATISTICAL ANALYSIS (CALCULATED BELOW LOWEST EVALUATION EDGE)
    % ==================================================================
    % Anchor to the true bottom edge of the two evaluation blocks to guarantee a clear visual gap
    \coordinate (evalBottom) at ($ (evalSynthesized.south)!0.5!(evalBaselines.south) $);

    \node[
        wideNode,
        fill=purple!10,
        below=1.15cm of evalBottom
    ] (analysis) {
        \textbf{Statistical Analysis and Comparison} \\[2pt]
        \scriptsize
        Empirical convergence trajectories, empirical runtime distributions (ECDF), and reference benchmark comparisons
    };

    % ==================================================================
    % 8. FINAL RESULTS
    % ==================================================================
    \node[
        wideNode,
        fill=green!15,
        below=0.85cm of analysis
    ] (results) {
        \textbf{Experimental Results Analysis} \\[2pt]
        \scriptsize
        Comparative assessment against CMA-ES, DE, and PSO to address core research questions
    };

    % ==================================================================
    % PIPELINE CONNECTORS
    % ==================================================================
    \draw[arrow] (problem) -- (promptScaffolding);
    \draw[arrow] (promptScaffolding) -- (synthesis);
    \draw[arrow] (synthesis) -- (champions);
    \draw[arrow] (champions) -- (benchmarkConfig);

    % Dual Evaluation Forks from Configuration Resolution
    \draw[branchArrow] (benchmarkConfig.south) -- ++(0,-0.28) -| (evalSynthesized.north);
    \draw[branchArrow] (benchmarkConfig.south) -- ++(0,-0.28) -| (evalBaselines.north);

    % Dual Evaluation Convergence into Analysis (adjusted drop clearance)
    \draw[branchArrow] (evalSynthesized.south) -- ++(0,-0.4) -| ([xshift=-3.0cm]analysis.north);
    \draw[branchArrow] (evalBaselines.south)   -- ++(0,-0.4) -| ([xshift=3.0cm]analysis.north);

    \draw[arrow] (analysis) -- (results);

    % ==================================================================
    % STAGE GROUPING CONTAINERS (BACKGROUND)
    % ==================================================================
    \begin{scope}[on background layer]
        \node[
            draw=blue!50!black,
            dashed,
            fill=blue!2,
            rounded corners=6pt,
            inner sep=9pt,
            fit=(synthesis) (champions),
            label={[font=\scriptsize\bfseries, text=blue!60!black]left:Synthesis Phase}
        ] (synthesisGroup) {};

        \node[
            draw=blue!50!black,
            dashed,
            fill=blue!2,
            rounded corners=6pt,
            inner sep=9pt,
            fit=(benchmarkConfig) (evalSynthesized) (evalBaselines),
            label={[font=\scriptsize\bfseries, text=blue!60!black]left:Benchmarking Phase}
        ] (benchmarkGroup) {};

        \node[
            draw=purple!60!black,
            dashed,
            fill=purple!2,
            rounded corners=6pt,
            inner sep=9pt,
            fit=(analysis) (results),
            label={[font=\scriptsize\bfseries, text=purple!70!black]left:Analysis Phase}
        ] (analysisGroup) {};
    \end{scope}

\end{tikzpicture}
